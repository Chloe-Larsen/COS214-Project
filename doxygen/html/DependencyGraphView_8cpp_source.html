102void DependencyGraphView::appendBatch()
103{
104 if (!nextScene) { batchTimer->stop(); return; }
105 for (int n = 0; n < 100 && nodeOffset < building.components.size(); ++n, ++nodeOffset) {
106 const ComponentRecord &c = building.components[nodeOffset]; const auto point = positions.positions.find(c.id);
107 if (point == positions.positions.end()) continue;
108 std::unique_ptr<NodeItem> item(new NodeItem(c)); item->setPos(point->second.first, point->second.second);
109 nextNodes.insert(QString::fromStdString(c.id), item.get()); nextScene->addItem(item.get()); item.release();
110 }
111 if (nodeOffset < building.components.size()) return;
112 for (int n = 0; n < 200 && edgeOffset < building.dependencies.size(); ++n, ++edgeOffset) {
113 const DependencyRecord &d = building.dependencies[edgeOffset];
114 const auto a = positions.positions.find(d.prerequisiteId), b = positions.positions.find(d.dependentId);
115 if (a == positions.positions.end() || b == positions.positions.end()) continue;
116 const QPointF start(a->second.first + 264, a->second.second + 49), end(b->second.first, b->second.second + 49);
117 const qreal bend = std::max(36.0, (end.x() - start.x()) * .5);
118 QPainterPath path(start); path.cubicTo(start + QPointF(bend, 0), end - QPointF(bend, 0), end);
119 std::unique_ptr<EdgeItem> edge(new EdgeItem(path)); edge->setData(0, "edge");
120 edge->setData(1, QString::fromStdString(d.prerequisiteId)); edge->setData(2, QString::fromStdString(d.dependentId));
121 edge->setToolTip("Prerequisite → dependent. Right-click to remove.");

124 nextScene->addItem(edge.get()); edge.release();
