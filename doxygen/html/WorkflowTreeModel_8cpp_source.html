17 QHash<QString, int> ids;

23 if (id.isEmpty() || ids.contains(id)) throw std::runtime_error("Engine snapshot contains an empty or duplicate component ID.");
24 ids.insert(id, i);
25 }

30 } else {

32 if (parent < 0 || !snapshot.components[parent].group)
33 throw std::runtime_error("Engine snapshot contains an unknown or non-group parent.");
34 nextParents[i] = parent;
35 nextPositions[i] = nextChildren[parent].size(); nextChildren[parent].push_back(i);
36 }
37 }

39 QVector<int> depth(count, 0);
40 for (int offset = 0; offset < order.size(); ++offset) {
41 const int id = order[offset];
42 if (depth[id] > 256) throw std::runtime_error("The displayed hierarchy exceeds 256 levels.");
43 for (int child : nextChildren[id]) { depth[child] = depth[id] + 1; order.push_back(child); }
44 }
45 if (order.size() != count) throw std::runtime_error("Engine snapshot contains a hierarchy cycle.");

47 std::vector<Stats> stats(count);
48 QVector<int> priorities(count);
49 QVector<QString> nextSummaries(count), nextAssignees(count);
50 QHash<QString, QString> names;
51 for (const WorkerRecord &person : snapshot.workers) names.insert(q(person.id), q(person.name));
52 for (int n = order.size() - 1; n >= 0; --n) {

54 const ComponentRecord &node = snapshot.components[i];
55 Stats &s = stats[i];
56 if (!node.group) { s.leaves = 1; s.complete = node.state == "COMPLETED" ? 1 : 0; s.priority = node.priority; s.workers.insert(node.workerId); }
57 else for (int child : nextChildren[i]) {
58 const Stats &other = stats[child];
59 s.leaves += other.leaves; s.complete += other.complete;
60 s.priority = std::max(s.priority, other.priority); s.workers.insert(other.workers.begin(), other.workers.end());
61 }
62 priorities[i] = s.priority == std::numeric_limits<int>::min() && node.group && s.leaves == 0 ? node.priority : s.priority;
63 nextSummaries[i] = node.group ? QString("%1/%2 complete").arg(s.complete).arg(s.leaves) : QString();
64 if (s.workers.size() > 1) nextAssignees[i] = "Multiple";
65 else if (s.workers.empty() || s.workers.begin()->empty()) nextAssignees[i] = "Unassigned";
66 else nextAssignees[i] = names.value(q(*s.workers.begin()), "Unknown worker");
67 }
68 beginResetModel();
69 current = snapshot; byId = ids; parents = nextParents; positions = nextPositions;
70 roots = nextRoots; children = nextChildren; displayPriority = priorities;
71 summaries = nextSummaries; assignees = nextAssignees; workerNames = names;
72 endResetModel();
73}

168bool WorkflowFilterModel::matches(const QModelIndex &idx) const
169{
170 if (state == "REJECTED") { if (!idx.data(WorkflowTreeModel::RejectedRole).toBool()) return false; }
171 else if (state == "ESCALATED") { if (!idx.data(WorkflowTreeModel::EscalatedRole).toBool()) return false; }
172 else if (!state.isEmpty() && idx.data(WorkflowTreeModel::StateRole).toString() != state) return false;
173 const QString assigned = idx.data(WorkflowTreeModel::WorkerRole).toString();
174 if (unassignedOnly && (!assigned.isEmpty() || idx.data(WorkflowTreeModel::GroupRole).toBool())) return false;
175 if (!worker.isEmpty() && assigned != worker) return false;
176 const int priority = idx.data(WorkflowTreeModel::PriorityRole).toInt();
177 if (priorityEnabled && (priority < minimum || priority > maximum)) return false;
178 return text.isEmpty() || idx.data().toString().contains(text, Qt::CaseInsensitive)
179 || idx.sibling(idx.row(), 2).data().toString().contains(text, Qt::CaseInsensitive);
180}
