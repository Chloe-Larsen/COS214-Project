256bool MainWindow::supported(Operation op) const
257{
258 if (modelFault || controller->busy() || !controller->snapshot().capabilities.count(op)) return false;
259 if (op == Operation::EditWorker || op == Operation::RemoveWorker || op == Operation::SetSupervisor) return pages->currentIndex() == 1 && !selectedWorker().isEmpty();
260 const ComponentRecord *node = treeModel->component(selection);
261 if (op == Operation::SetPriority && node && node->group) return false;
262 if (std::find(nodeOperations.begin(), nodeOperations.end(), op) != nodeOperations.end()) return pages->currentIndex() == 0 && node && node->allowedActions.count(op);
263 if ((op == Operation::AddItem || op == Operation::AddGroup) && node) {
264 const ComponentRecord *parent = node->group ? node : treeModel->component(q(node->parentId));
265 return !parent || parent->allowedActions.count(op);
266 }
267 return true;
268}
