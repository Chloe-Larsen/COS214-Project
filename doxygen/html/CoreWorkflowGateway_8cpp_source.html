359 void validateDependencies() const {
360 const std::size_t count = document.data.components.size();
361 std::vector<std::vector<std::size_t> > edges(count * 2);
362 std::vector<int> incoming(count * 2, 0);
363 const auto link = [&](std::size_t a, std::size_t b) { edges[a].push_back(b); ++incoming[b]; };
364 for (std::size_t i = 0; i < count; ++i) {
365 const ComponentRecord &c = document.data.components[i];
366 const auto list = children.find(c.id);
367 if (!c.group || list == children.end() || list->second.empty()) link(i * 2, i * 2 + 1);
368 else for (std::size_t child : list->second) { link(i * 2, child * 2); link(child * 2 + 1, i * 2 + 1); }
369 }
370 // Start/finish barriers include group membership in cycle detection.
371 // No Cartesian expansion of all descendant leaf pairs is needed.
372 for (const DependencyRecord &d : document.data.dependencies)
373 link(indexes.at(d.prerequisiteId) * 2 + 1, indexes.at(d.dependentId) * 2);
374 std::queue<std::size_t> ready;
375 for (std::size_t i = 0; i < incoming.size(); ++i) if (incoming[i] == 0) ready.push(i);
376 std::size_t visited = 0;
377 while (!ready.empty()) {
378 const std::size_t i = ready.front(); ready.pop(); ++visited;
379 for (std::size_t next : edges[i]) if (--incoming[next] == 0) ready.push(next);
380 }
381 require(visited == incoming.size(), "/dependencies", "This dependency creates a cycle, including an item/group relationship.");

507 std::vector<std::string> result, pending(1, id);
508 for (std::size_t i = 0; i < pending.size(); ++i) {
509 const ComponentRecord &c = document.data.components.at(indexes.at(pending[i]));
510 if (!c.group) result.push_back(c.id);
511 const auto list = children.find(c.id);
512 if (list != children.end()) for (std::size_t child : list->second) pending.push_back(document.data.components[child].id);
513 }
514 return result;
515 }
516 std::size_t assignUnassigned(const std::string &only = "") {
517 std::vector<Worker *> workers;
518 std::vector<Item *> items;
519 for (const auto &w : people.values) workers.push_back(w.second.get());
520 for (const auto &entry : components)
521 if (Item *item = dynamic_cast<Item *>(entry.second)) items.push_back(item);
522 std::stable_sort(items.begin(), items.end(), [](Item *a, Item *b) {
523 return a->getPriority() > b->getPriority();
524 });
525 std::unique_ptr<AssignmentStrategy> policy;
526 if (document.data.assignmentStrategy == "RoleBased") policy.reset(new GuiRoleBased(workers, items, document.data.strategyOptions));
527 else if (document.data.assignmentStrategy == "WorkloadBased") policy.reset(new GuiWorkloadBased(workers, items, document.data.strategyOptions));
528 if (!policy) return 0;
529 Item *selected = only.empty() ? nullptr : dynamic_cast<Item *>(component(only));
530 std::size_t assigned = 0;
531 for (Item *item : items) if (!item->getWorker() && item->getStatus() != "COMPLETED" && item->getStatus() != "CANCELLED") {
532 if (selected && item != selected) continue;
533 Worker *chosen = policy->assign(item, nullptr);
534 if (chosen) { Assign assign(item, chosen); assign.execute(); ++assigned; }
535 }
536 refresh();
537 return assigned;
538 }
539 void act(const OperationRequest &request) {
540 const WorkflowSnapshot s = snapshot(0);
541 const auto selected = std::find_if(s.components.begin(), s.components.end(),
542 [&](const ComponentRecord &c) { return c.id == request.targetId; });
543 require(selected != s.components.end() && selected->allowedActions.count(request.operation),
544 "/operation", "This action is not available for the selected work and state.");
545 if (request.operation == Operation::RenameComponent) {
546 component(request.targetId)->setName(request.text); refresh(); return;
547 }
548 const std::vector<std::string> targets = leaves(request.targetId);
549 require(!targets.empty(), "/componentId", "This group has no items.");
550 std::size_t performed = 0;
551 for (const std::string &id : targets) {
552 Item *item = dynamic_cast<Item *>(component(id));
553 if (selected->group) {
554 const ComponentRecord &leaf = *std::find_if(s.components.begin(), s.components.end(),
555 [&](const ComponentRecord &c) { return c.id == id; });
556 if (!leaf.allowedActions.count(request.operation)) continue;
557 }
558 std::unique_ptr<WorkflowCommand> command;
559 switch (request.operation) {
560 case Operation::SetPriority: item->setPriority(request.component.priority); break;
561 case Operation::Assign: command.reset(new Assign(item, worker(request.relatedId))); break;
562 case Operation::Unassign: item->assignWorker(nullptr); break;
563 case Operation::Progress: command.reset(new Progress(item)); break;
564 case Operation::Review:
565 if (item->getStatus() != "IN_PROGRESS" && selected->group) continue;
566 require(request.options.count("decision") &&
567 (option(request.options, "decision") == "approve" || option(request.options, "decision") == "reject"),
568 "/review", "Choose Approve or Reject in the review dialog.");
569 require(option(request.options, "decision") != "reject" || !request.text.empty(), "/review/reason", "Enter a rejection reason.");
570 if (item->getWorker() && item->getWorker()->getSupervisor())
571 item->getWorker()->getSupervisor()->setDecisionPolicy([request](Item *) { return option(request.options, "decision") == "approve"; });
572 item->requestReview();
573 if (option(request.options, "decision") == "reject") item->reject(request.text);
574 break;
575 case Operation::Reject:
576 require(!request.text.empty(), "/reason", "Enter a rejection reason.");
577 command.reset(new Reject(item, request.text)); break;
578 case Operation::Escalate: command.reset(new Escalate(item)); break;
579 case Operation::Cancel:
580 if (item->getStatus() == "COMPLETED" || item->getStatus() == "CANCELLED") continue;
581 command.reset(new Cancel(item)); break;
582 default: throw InputError("/operation", "Unsupported item action.");
583 }
584 if (command) command->execute();
585 ++performed;
586 }
587 require(performed != 0, "/state", "No items in this group can perform the selected action.");
588 syncDependencies(); refresh();
589 }

619
620OperationResult CoreWorkflowGateway::execute(const OperationRequest &request)
621{
622 try {
623 WorkflowDocument document = state->document;
624 const auto worker = [&](const std::string &id) -> WorkerRecord & {
625 for (WorkerRecord &w : document.data.workers) if (w.id == id) return w;
626 throw InputError("/workerId", "Unknown worker: " + id);
627 };
628 const auto component = [&](const std::string &id) -> ComponentRecord & {
629 for (ComponentRecord &c : document.data.components) if (c.id == id) return c;
630 throw InputError("/componentId", "Unknown component: " + id);
631 };
632 bool itemAction = false, automatic = false;
633 switch (request.operation) {
634 case Operation::CreateWorkflow:
635 document = WorkflowDocument(); document.data.name = request.text; break;
636 case Operation::AddWorker: document.data.workers.push_back(request.worker); break;
637 case Operation::EditWorker: {
638 require(request.worker.id.empty() || request.worker.id == request.targetId, "/worker/id", "A worker ID cannot be changed.");
639 WorkerRecord &existing = worker(request.targetId);
640 existing = request.worker; existing.id = request.targetId; break;
641 }
642 case Operation::SetSupervisor: worker(request.targetId).supervisorId = request.relatedId; break;
643 case Operation::RemoveWorker: {
644 worker(request.targetId);
645 for (const WorkerRecord &w : document.data.workers)
646 require(w.supervisorId != request.targetId, "/workerId", "Change this worker's subordinate relationships before removal.");
647 for (const ComponentRecord &c : document.data.components)
648 require(c.workerId != request.targetId, "/workerId", "Unassign this worker's items before removal.");
649 auto &workers = document.data.workers;
650 workers.erase(std::remove_if(workers.begin(), workers.end(), [&](const WorkerRecord &w) { return w.id == request.targetId; }), workers.end());
651 break;
652 }
653 case Operation::AddItem:
654 case Operation::AddGroup: {
655 ComponentRecord c = request.component;
656 c.group = request.operation == Operation::AddGroup;
657 c.state = c.group ? "" : "AVAILABLE"; c.rejected = c.escalated = false;
658 c.rejectionReason.clear(); c.allowedActions.clear();
659 if (c.group) { c.workerId.clear(); c.deadline.clear(); }
660 document.data.components.push_back(c); automatic = !c.group; break;
661 }
662 case Operation::SetStrategy:
663 document.data.assignmentStrategy = request.text; document.data.strategyOptions = request.options; automatic = true; break;
664 case Operation::AddDependency:
665 case Operation::RemoveDependency: {
666 ComponentRecord &dependent = component(request.targetId); component(request.relatedId);
667 for (const std::string &id : state->leaves(dependent.id)) {
668 const std::string status = component(id).state;
669 require(status == "AVAILABLE" || status == "BLOCKED", "/dependencies", "Change dependencies before starting the dependent work.");
670 }
671 auto &links = document.data.dependencies;
672 if (request.operation == Operation::AddDependency) {
673 DependencyRecord edge; edge.prerequisiteId = request.relatedId; edge.dependentId = request.targetId; links.push_back(edge);
674 } else {
675 const auto it = std::find_if(links.begin(), links.end(), [&](const DependencyRecord &d) {
676 return d.prerequisiteId == request.relatedId && d.dependentId == request.targetId;
677 });
678 require(it != links.end(), "/dependencies", "The selected dependency no longer exists."); links.erase(it);
679 }
680 break;
681 }
682 case Operation::ConfigureNotifier:
683 case Operation::ConfigureAudit: {
684 const bool notifier = request.operation == Operation::ConfigureNotifier;
685 validateObserver(request.options, notifier);
686 if (request.targetId.empty()) {
687 const std::string prefix = notifier ? "notifier." : "audit.";
688 for (auto it = document.engineData.begin(); it != document.engineData.end(); )
689 if (it->first.compare(0, prefix.size(), prefix) == 0) it = document.engineData.erase(it); else ++it;
690 for (const auto &entry : request.options) document.engineData[prefix + entry.first] = entry.second;
691 } else {
692 ComponentRecord &c = component(request.targetId);
693 (notifier ? c.notifier : c.audit) = request.options;
694 }
695 break;
696 }
697 default:
698 component(request.targetId);
699 if (request.operation == Operation::Review)
700 for (const std::string &id : state->leaves(request.targetId)) {
701 component(id).rejected = false; component(id).rejectionReason.clear();
702 }
703 itemAction = true; break;
704 }
705 std::unique_ptr<Storage> next(new Storage(document));
706 if (itemAction) next->act(request);
707 const std::size_t assigned = automatic ? next->assignUnassigned(
708 request.operation == Operation::AddItem ? request.component.id : "") : 0;
709 require(revision != std::numeric_limits<std::uint64_t>::max(), "/revision", "The workflow revision is exhausted.");
710 OperationResult result; result.ok = true;
711 result.selectedId = request.operation == Operation::AddItem || request.operation == Operation::AddGroup ? request.component.id : request.targetId;
712 result.message = operationName(request.operation) + " completed.";
713 if (request.operation == Operation::SetStrategy)
714 result.message = request.text.empty() ? "Manual assignment selected."
715 : "Assignment strategy applied: " + std::to_string(assigned) + " item(s) assigned.";
716 if (request.operation == Operation::Progress || request.operation == Operation::Review) {
717 const ComponentRecord &c = next->record(request.targetId);
718 if (!c.group) result.message = c.name + ": " + (c.rejected ? "Review rejected." : c.state + ".");
719 }
720 next->append("System", result.message, result.selectedId);
721 // Commit before invoking output adapters. A delivery failure must never
722 // report an already-committed workflow action as rolled back.
723 state.swap(next); ++revision;
724 try {
725 for (const ComponentRecord &c : state->document.data.components) {
726 if (c.group) continue;
727 const auto before = next->indexes.find(c.id);
728 const ComponentRecord *old = before == next->indexes.end() ? nullptr : &next->document.data.components[before->second];
729 if ((!old && !c.workerId.empty()) || (old && old->workerId != c.workerId))
730 state->publish("ASSIGNED", c.id, c.workerId.empty() ? c.name + " was unassigned." : c.name + " assigned to " + state->worker(c.workerId)->getName() + ".");
731 if (old && c.state != old->state)
732 state->publish(c.state, c.id, c.name + ": " + old->state + " → " + c.state + ".");
733 if (old && c.rejected && (!old->rejected || c.rejectionReason != old->rejectionReason))
734 state->publish("REJECTED", c.id, c.name + ": " + c.rejectionReason);
735 if (old && c.escalated && !old->escalated)
736 state->publish("ESCALATED", c.id, c.name + " was escalated.");
737 }
738 state->publish("UPDATED", state->components.count(result.selectedId) ? result.selectedId : "", result.message);
739 } catch (...) { result.message += " Some activity output could not be recorded."; }
740 if (invalidated) { try { invalidated(); } catch (...) {} }
741 return result;
742 } catch (const DocumentError &e) {
743 OperationResult result; result.message = "The workflow was not changed."; result.diagnostics = e.diagnostics(); return result;
744 } catch (const InputError &e) { return failed(e.path, e.what()); }
745 catch (const std::exception &e) { return failed("/operation", e.what()); }
746 catch (const std::string &e) { return failed("/operation", e); }
747 catch (const char *e) { return failed("/operation", e ? e : "The core rejected the action."); }
748 catch (...) { return failed("/operation", "The operation could not be completed."); }
749}
