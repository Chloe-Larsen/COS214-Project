29static std::time_t parseItemDeadline(const std::string &text)
30{
31 if (text.empty())
32 return 0;
33 if (text.size() != 10 && text.size() != 19)
34 throw std::string("Use YYYY-MM-DD for the deadline.");
35
36 for (std::size_t i = 0; i < text.size(); ++i)
37 {
38 char separator = 0;
39 if (i == 4 || i == 7)
40 separator = '-';
41 if (i == 10)
42 separator = ' ';
43 if (i == 13 || i == 16)
44 separator = ':';
45 if ((separator && text[i] != separator) ||
46 (!separator && (text[i] < '0' || text[i] > '9')))
47 throw std::string("Invalid deadline: " + text);
48 }
49
50 std::tm date = {};
51 int year = 0;
52 for (int i = 0; i < 4; ++i)
53 year = year * 10 + text[i] - '0';
54 date.tm_year = year - 1900;
55 date.tm_mon = (text[5] - '0') * 10 + text[6] - '0' - 1;
56 date.tm_mday = (text[8] - '0') * 10 + text[9] - '0';
57 if (text.size() == 19)
58 {
59 date.tm_hour = (text[11] - '0') * 10 + text[12] - '0';
60 date.tm_min = (text[14] - '0') * 10 + text[15] - '0';
61 date.tm_sec = (text[17] - '0') * 10 + text[18] - '0';
62 }
63 date.tm_isdst = -1;
64 const std::tm requested = date;
65 std::time_t result = std::mktime(&date); // mktime normalises dates
66 if (result == static_cast<std::time_t>(-1) ||
67 date.tm_year != requested.tm_year || date.tm_mon != requested.tm_mon ||
68 date.tm_mday != requested.tm_mday || date.tm_hour != requested.tm_hour ||
69 date.tm_min != requested.tm_min || date.tm_sec != requested.tm_sec)
70 throw std::string("Invalid deadline: " + text);
71 return result;
72}
